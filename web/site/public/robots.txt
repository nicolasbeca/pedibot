# One group on purpose. Naming a crawler here starts a NEW group for it, and a named group does
# not inherit the rules below — so adding "User-agent: GPTBot / Allow: /" without repeating the
# Disallow lines would hand that crawler the private share links under /a/. The answer engines
# (GPTBot, OAI-SearchBot, ClaudeBot, PerplexityBot and the rest) are already welcome through this
# group, with the same three exceptions as everybody else. Leave it as one group.
User-agent: *
Allow: /

# not for search engines: the API, the private share links and the admin panel
Disallow: /api/
Disallow: /a/
Disallow: /admin

Sitemap: https://pedibot.xyz/sitemap-index.xml
